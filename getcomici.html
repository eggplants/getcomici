getcomici

getcomici

Retrieve and save images from manga distribution sites using Comici+.

Valid URL Formats

  • <host>/episodes/<id>
    • e.g. https://mangabu.jp/episodes/71f48a2c352ed

212    def get(  # noqa: PLR0913
213        self,
214        url: str,
215        save_path: str | Path = ".",
216        *,
217        overwrite: bool = False,
218        only_first: bool = False,
219        save_metadata: bool = False,
220        print_log: bool = False,
221    ) -> tuple[str | None, Path, bool]:
222        """Download one episode and unscramble every page of it.
223
224        Args:
225            url: The episode URL.
226            save_path: Directory to build `<series>/<episode>/` under.
227            overwrite: Download again even if the directory already exists.
228            only_first: Stop after the first page.
229            save_metadata: Also write the raw `contentsInfo` response.
230            print_log: Draw a progress bar.
231
232        Returns:
233            The next episode's URL (or None), the directory written, and
234            whether anything was actually downloaded.

236        episode = self.episode_info(url)
237        save_dir = Path(save_path) / sanitize_filename(episode.series_title) / sanitize_filename(episode.episode_title)
238        if save_dir.exists() and not overwrite:
239            return episode.next_url, save_dir, False
240
241        pages = self.pages(episode)
242        if not pages:
243            warnings.warn(episode.episode_title, NeedPurchase, stacklevel=2)
244            return episode.next_url, save_dir, False
245
246        save_dir.mkdir(parents=True, exist_ok=True)
247        if save_metadata:
248            (save_dir / "metadata.json").write_text(
249                json.dumps(pages, indent=4, ensure_ascii=False),
250                encoding="utf-8",
251            )
252        self._save_pages(
253            pages,
254            save_dir,
255            referer=episode.url,
256            only_first=only_first,
257            print_log=print_log,
258        )
259        return episode.next_url, save_dir, True

261    def login(self, url: str, user_id: str, password: str) -> None:
262        """Sign in, so episodes the account may read become readable.
263
264        Sites run NextAuth behind `/api/auth`, with a credentials provider that
265        takes a Comici ID or an email address. The session cookie lands on the
266        shared session and the returned id token is sent with later API calls.
267        This grants nothing the account does not already own.
268
269        Args:
270            url: Any URL on the site to sign in to.
271            user_id: A Comici ID or the email address the account uses.
272            password: The account's password.
273
274        Raises:
275            LoginError: The site refused the credentials.
276        """
277        origin = self._origin(url)
278        auth = f"{origin}/api/auth"
279        csrf = self._session.get(f"{auth}/csrf", headers=HEADERS, timeout=30)
280        csrf.raise_for_status()
281
282        res = self._session.post(
283            f"{auth}/callback/credentials",
284            data={
285                "id": user_id,
286                "password": password,
287                "csrfToken": csrf.json()["csrfToken"],
288                "callbackUrl": f"{origin}/",
289                "json": "true",
290            },
291            headers=HEADERS,
292            timeout=30,
293        )
294        res.raise_for_status()
295
296        session = self._session.get(f"{auth}/session", headers=HEADERS, timeout=30)
297        session.raise_for_status()
298        token = (session.json() or {}).get("idToken")
299        if not token:
300            msg = f"{origin} refused the credentials for {user_id!r}."
301            raise LoginError(msg)
302        self._id_tokens[origin] = str(token)

306        parsed = urlparse(url)
307        return f"{parsed.scheme}://{parsed.netloc}"

309    def _headers(self, url: str) -> dict[str, str]:
310        """Headers for `url`, carrying the id token once its site is signed in."""
311        token = self._id_tokens.get(self._origin(url))
312        return {**HEADERS, "Authorization": token} if token else dict(HEADERS)

314    def episode_info(self, url: str) -> Episode:
315        """Read the viewer parameters off an episode page.
316
317        Args:
318            url: The episode URL.
319
320        Returns:
321            The parsed episode.
322
323        Raises:
324            NotAComiciPageError: The page carries no Comici+ viewer.
325        """
326        res = self._session.get(url, headers=self._headers(url), timeout=30)
327        res.raise_for_status()
328        soup = BeautifulSoup(res.content, "html.parser")
329
330        viewer = soup.find(id=_VIEWER_ID)
331        if not isinstance(viewer, Tag):
332            return self._episode_from_api(url)
333
334        viewer_id = str(viewer.attrs.get("data-comici-viewer-id", "")) or None
335        if viewer_id is None:
336            msg = f"'#{_VIEWER_ID}' on {url} carries no data-comici-viewer-id."
337            raise NotAComiciPageError(msg)
338
339        api_domain = str(viewer.attrs.get("data-api-domain", "/api"))
340        parsed = urlparse(url)
341        api_base = (
342            f"{parsed.scheme}://{parsed.netloc}{api_domain}" if api_domain.startswith("/") else f"https://{api_domain}"
343        )
344
345        series_title, episode_title = self._titles(soup, viewer, viewer_id)
346
347        next_id = str(viewer.attrs.get("data-next-episode-id", ""))
348        next_url = urljoin(url, next_id) if next_id else None
349
350        return Episode(
351            url=url,
352            viewer_id=viewer_id,
353            member_jwt=str(viewer.attrs.get("data-member-jwt", "")),
354            api_base=api_base,
355            series_title=series_title,
356            episode_title=episode_title,
357            next_url=next_url,
358        )

360    def pages(self, episode: Episode, member_jwt: str | None = None) -> list[Page]:
361        """List every page of an episode.
362
363        `contentsInfo` refuses a range wider than the episode, so the total is
364        asked for first and the real range fetched second.
365
366        Args:
367            episode: The episode to list.
368            member_jwt: Overrides the token the episode page carried, if any.
369
370        Returns:
371            The pages, in reading order. Empty when the episode is not readable.
372        """
373        if episode.inline_pages is not None:
374            return list(episode.inline_pages)
375        if member_jwt is None:
376            member_jwt = episode.member_jwt
377        total = int(self._contents_info(episode, 0, 0, member_jwt).get("totalPages") or 0)
378        if total <= 0:
379            return []
380        body = self._contents_info(episode, 0, total - 1, member_jwt)
381        pages: list[Page] = list(body.get("result") or [])
382        return sorted(pages, key=lambda page: page["sort"])

384    def _episode_from_api(self, url: str) -> Episode:
385        """Read an episode that renders its viewer only after hydration.
386
387        Newer sites (ebookstore.corkagency.com) ship an episode page with no
388        `#comici-viewer` element on it, and their `/api/episodes/{id}` hands the
389        page images over directly, already unscrambled, instead of a viewer id
390        to look up with `contentsInfo`.
391
392        Args:
393            url: The episode URL.
394
395        Returns:
396            The parsed episode, carrying its pages.
397
398        Raises:
399            NotAComiciPageError: The API describes no episode either.
400        """
401        parsed = urlparse(url)
402        episode_id = parsed.path.rstrip("/").rsplit("/", 1)[-1]
403        api_base = f"{parsed.scheme}://{parsed.netloc}/api"
404        res = self._session.get(
405            f"{api_base}/episodes/{episode_id}",
406            headers={**self._headers(url), "Referer": url},
407            timeout=30,
408        )
409        body = res.json() if res.ok else None
410        episode = body.get("episode") if isinstance(body, dict) else None
411        if not isinstance(episode, dict):
412            msg = f"no '#{_VIEWER_ID}' element on {url}, and its episode API describes none either."
413            raise NotAComiciPageError(msg)
414
415        series = episode.get("series") or {}
416        summary = episode.get("summary") or {}
417        next_id = str(episode.get("nextEpisodeId") or "")
418        return Episode(
419            url=url,
420            viewer_id=str(episode.get("id") or episode_id),
421            api_base=api_base,
422            series_title=str(series.get("name") or "").strip() or episode_id,
423            episode_title=str(summary.get("title") or "").strip() or episode_id,
424            next_url=urljoin(url, next_id) if next_id else None,
425            inline_pages=self._inline_pages(episode),
426        )

428    @staticmethod
429    def _inline_pages(episode: dict[str, Any]) -> tuple[Page, ...]:
430        """Turn the `content` blocks of an episode JSON into pages."""
431        return tuple(
432            Page(
433                imageUrl=str(node["url"]),
434                scramble=_IDENTITY_SCRAMBLE,
435                sort=index,
436                width=int(node.get("width") or 0),
437                height=int(node.get("height") or 0),
438                expiresOn=0,
439            )
440            for index, node in enumerate(
441                node
442                for node in episode.get("content") or []
443                if isinstance(node, dict) and node.get("type") == "image" and node.get("url")
444            )
445        )

447    def _contents_info(
448        self,
449        episode: Episode,
450        page_from: int,
451        page_to: int,
452        member_jwt: str = "",
453    ) -> dict[str, Any]:
454        res = self._session.get(
455            f"{episode.api_base}/book/contentsInfo",
456            params={
457                "user-id": member_jwt,
458                "comici-viewer-id": episode.viewer_id,
459                "page-from": page_from,
460                "page-to": page_to,
461            },
462            # Some sites (studio.booklista.co.jp) answer 403 without a site Referer.
463            headers={**self._headers(episode.url), "Referer": episode.url},
464            timeout=30,
465        )
466        res.raise_for_status()
467        body = res.json()
468        if not isinstance(body, dict) or "result" not in body:
469            msg = f"contentsInfo refused the request: {body}"
470            raise ComiciError(msg)
471        return body

473    @staticmethod
474    def _titles(soup: BeautifulSoup, viewer: Tag, viewer_id: str) -> tuple[str, str]:
475        """Split `og:title` -- `"<series>・<episode> | <site>"` -- into its parts."""
476        og = soup.find("meta", property="og:title")
477        heading = str(og.attrs.get("content", "")) if isinstance(og, Tag) else ""
478        if not heading and soup.title:
479            heading = soup.title.get_text()
480        heading = heading.rsplit(" | ", 1)[0].strip()
481
482        series, _, episode = heading.partition("・")
483        if not episode:
484            series, episode = str(viewer.attrs.get("data-share-text", "")), heading
485        return (series.lstrip("#").strip() or viewer_id, episode.strip() or viewer_id)

487    def _save_pages(
488        self,
489        pages: list[Page],
490        save_dir: Path,
491        *,
492        referer: str,
493        only_first: bool = False,
494        print_log: bool = False,
495    ) -> None:
496        wanted = pages[:1] if only_first else pages
497        width = len(str(len(wanted)))
498        progress = Progress(
499            SpinnerColumn(),
500            TextColumn("{task.description}"),
501            BarColumn(),
502            TaskProgressColumn(),
503            TextColumn("("),
504            MofNCompleteColumn(),
505            TextColumn("pages )"),
506            TextColumn("remain:"),
507            TimeRemainingColumn(),
508            TextColumn("spent:"),
509            TimeElapsedColumn(),
510            disable=not print_log,
511        )
512        with progress:
513            task = progress.add_task("[red]Downloading...", total=len(wanted))
514            for index, page in enumerate(wanted):
515                image = self._image(page, referer)
516                image.save(save_dir / f"{index:0{width}d}.jpg", quality=95)
517                progress.update(task, advance=1)
518
519    def _image(self, page: Page, referer: str) -> Image.Image:
520        res = self._session.get(
521            page["imageUrl"],
522            headers={**self._headers(referer), "Referer": referer},
523            timeout=60,
524        )
525        res.raise_for_status()
526        image = Image.open(BytesIO(res.content))
527        return descramble(image, parse_scramble(page["scramble"]))

def descramble(image: PIL.Image.Image, scramble: Sequence[int]) -> PIL.Image.Image: 
146def descramble(image: Image.Image, scramble: Sequence[int]) -> Image.Image:
147    """Put a scrambled page back together.
148
149    The viewer walks the destination grid column by column and copies
150    `scramble[n]`-th source tile into the n-th destination slot. Tile size is
151    floored, so any leftover strip on the right and bottom edge is never
152    shuffled and is kept as-is.
153
154    Args:
155        image: The page exactly as the CDN serves it.
156        scramble: Source tile index per destination tile, from `parse_scramble`.
157
158    Returns:
159        A new image with the tiles back in reading order.
160    """
161    width, height = image.size
162    tile_width, tile_height = width // COLUMNS, height // ROWS
163    out = image.copy()
164    for dest, src in enumerate(scramble):
165        dest_col, dest_row = divmod(dest, ROWS)
166        src_col, src_row = divmod(src, ROWS)
167        tile = image.crop(
168            (
169                tile_width * src_col,
170                tile_height * src_row,
171                tile_width * (src_col + 1),
172                tile_height * (src_row + 1),
173            ),
174        )
175        out.paste(tile, (tile_width * dest_col, tile_height * dest_row))
176    return out
